# Base requirements for running this pack's skill examples and validators.
#
# Every `skills/*/example_*.py --dry-run` runs with just these. Skills that
# teach a third-party project declare their extra packages in
# requirements-extras.txt, grouped by skill.
#
#     pip install -r requirements.txt
#     uv run --with dspy python skills/<skill>/example_*.py --dry-run

# Floor is 3.3.0: DSPy 3.3.0 renamed dspy.RLM's `max_iterations` to
# `max_iters` and replaced `interpreter=` with `interpreter_factory=`,
# so skills/dspy-rlm-module cannot run on the 3.2.x line.
dspy>=3.3.0,<3.4
pytest>=8.0
